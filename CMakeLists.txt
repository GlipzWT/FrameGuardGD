cmake_minimum_required(VERSION 3.21)
project(FrameGuard VERSION 0.1.0 LANGUAGES CXX)
option(FRAMEGUARD_CORE_ONLY "Build standalone planner tests without Geode" OFF)
if(FRAMEGUARD_CORE_ONLY)
    enable_testing()
    add_executable(frameguard_tests tests/planner_tests.cpp)
    target_compile_features(frameguard_tests PRIVATE cxx_std_17)
    target_include_directories(frameguard_tests PRIVATE include)
    add_test(NAME planner COMMAND frameguard_tests)
    return()
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
if(NOT DEFINED ENV{GEODE_SDK})
    message(FATAL_ERROR "GEODE_SDK is missing. Install the Geode SDK / see README_RU.md")
endif()
add_library(${PROJECT_NAME} SHARED src/main.cpp src/Assist.cpp src/Panel.cpp)
target_include_directories(${PROJECT_NAME} PRIVATE include)
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
setup_geode_mod(${PROJECT_NAME})
